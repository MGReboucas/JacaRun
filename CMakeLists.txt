cmake_minimum_required(VERSION 3.22)
project(JacaRunCore LANGUAGES CXX)

add_library(jacarun_core src/GameManager.cpp src/Level.cpp src/Player.cpp src/Profile.cpp)
target_include_directories(jacarun_core PUBLIC include)
target_compile_features(jacarun_core PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(jacarun_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(jacarun_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(jacarun_cli main.cpp)
target_link_libraries(jacarun_cli PRIVATE jacarun_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(mechanics_tests tests/mechanics_tests.cpp)
  target_link_libraries(mechanics_tests PRIVATE jacarun_core)
  add_test(NAME mechanics COMMAND mechanics_tests)
endif()
